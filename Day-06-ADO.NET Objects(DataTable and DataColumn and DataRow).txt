6. DataTable In ADO.NET

* DataTable represents a single table in memory

What is the difference between DataSet and DataTable?

* A DataSet can contain multiple tables
* A DataTable contains only one table

Example :

using System.Data;
using System.Data.SqlClient;

namespace Ado.NET_Objects
{
    class Program
    {
        static void Main(string[] args)
        {
            string connectionString = "server=(localdb)\\MSSQLLocalDB;Database=DotNet11AmBatch;Integrated Security=true;";

            SqlConnection conn = new SqlConnection(connectionString);

            try
            {
                conn.Open();

                Console.WriteLine("Connection Opened Successfully");

                SqlDataAdapter da = new SqlDataAdapter("SELECT * FROM Employees", conn);

                DataTable dt = new DataTable();

                da.Fill(dt);

                foreach(DataRow dr in dt.Rows) 
                {
                    Console.WriteLine(dr["EmployeeId"] + " " + dr["EmployeeName"] + " " + dr["Salary"]);
                }
            }
            catch(SqlException ex)
            {
                Console.WriteLine("Database Error : ");
                Console.WriteLine("Reason : " + ex.Message);
            }
            catch (Exception ex)
            {
                Console.WriteLine("Unexpected Error : ");

                Console.WriteLine("Reason : " + ex.Message);
            }
            finally
            {
                if(conn.State == System.Data.ConnectionState.Open)
                {
                    conn.Close();

                    Console.WriteLine("Connection Closed");
                }
            }

            Console.ReadLine();
        }
    }
}

* DataRow represents a single row(record) in a DataTable.

	- DataTable is Entire Table
	- DataRow is one Record inside the table

* Creating a DataTable.	

	DataTable dt = new DataTable();
	
	dt.Columns.Add("EmployeeId");
	dt.Columns.Add("EmployeeName");
	dt.Columns.Add("Salary");
	
* Creating DataRow
		
	DataRow row = dt.NewRow();
	
* Assigning Values
		
	row["EmployeeId"] = 101;
	row["EmployeeName"] = "Rahul";
	row["Salary"] = 70000;

* Adding Row to DataTable
		
	dt.Rows.Add(row);

Result :

EmployeeId	EmployeeName	Salary
101			Rahul			70000.00

Example :

using System.Data;
using System.Data.SqlClient;

namespace Ado.NET_Objects
{
    class Program
    {
        static void Main(string[] args)
        {
            try
            {
                DataTable dt = new DataTable();

                dt.Columns.Add("EmployeeId");
                dt.Columns.Add("EmployeeName");
                dt.Columns.Add("Salary");

                DataRow row = dt.NewRow();

                row["EmployeeId"] = 101;
                row["EmployeeName"] = "Rahul";
                row["Salary"] = 70000;

                dt.Rows.Add(row);

                foreach (DataRow dr in dt.Rows)
                {
                    Console.WriteLine(dr["EmployeeId"] + " " + dr["EmployeeName"] + " " + dr["Salary"]);
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine("Error Occurred");
                Console.WriteLine(ex.Message);
            }
            finally
            {
                Console.WriteLine("Program Execution Completed");
            }

            Console.ReadLine();
        }
    }
}

using System.Data;
using System.Data.SqlClient;

namespace Ado.NET_Objects
{
    class Program
    {
        static void Main(string[] args)
        {
            try
            {
                DataTable dt = new DataTable();

                dt.Columns.Add("EmployeeId");
                dt.Columns.Add("EmployeeName");
                dt.Columns.Add("Salary");

                foreach (DataColumn col in dt.Columns)
                {
                    Console.WriteLine(col.ColumnName);
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine("Error Occurred");
                Console.WriteLine(ex.Message);
            }
            finally
            {
                Console.WriteLine("Program Execution Completed");

    }
