Loops In TypeScript
-------------------

1. For Loop

Syntax : for(initialization; condition; increment/decrement){}

Example 1 : Print numbers 1 to 5

for (let i = 1; i <= 5; i++) {
  console.log(i);
}

Example 2 : Employee Salary Processing

let employees = 5;

for (let employee = 1; employee <= employees; employee++) {
  let salary = employee * 10000;
  let experience = employee + 1;
  let department = employee;

  if (salary >= 50000) {
    console.log("Salary Level : High");
  } else if (salary >= 30000) {
    console.log("Salary Level : Medium");
  } else {
    console.log("Salary Level : Low");
  }

  if (experience >= 5) {
    console.log("Experience Level : Senior");
  } else if (experience >= 3) {
    console.log("Experience Level : Mid-Level");
  } else {
    console.log("Experience Level : Junior");
  }

  switch (department) {
    case 1:
      console.log("Department : HR");
      break;
    case 2:
      console.log("Department : Finance");
      break;
    case 3:
      console.log("Department : IT");
      break;
    case 4:
      console.log("Department : Marketing");
      break;
    case 5:
      console.log("Department : Sales");
      break;
    default:
      console.log("Department : Unknown");
  }

  let bonus = 0;

  if (salary >= 50000) {
    bonus = (salary * 20) / 100;
  } else if (salary >= 30000) {
    bonus = (salary * 10) / 100;
  } else {
    bonus = (salary * 5) / 100;
  }

  console.log("Bonus : ", bonus);

  let finalSalary = salary + bonus;

  console.log("Final Salary : ", finalSalary);

  console.log("==================================");
}

2. While Loop

Example 1 : Print numbers 1 to 3

let i = 1;

while (i <= 3) {
  console.log(i);
  i++;
}

3. Do-While Loop

Example 1 : Print numbers 1 to 5

let i = 1;

do {
  console.log(i);
  i++;
} while (i <= 5);

4. ForEach Loop

Syntax :

array.forEach(item) => {
	// code to execute
});

Example : 

let names = ["Rahul", "Priya", "Arjun"];

names.forEach((name) => {
  console.log(name);
});
