Connected and Disconnected Architectures
-----------------------------------------

1. Connected Architecture

* In Connected Architecture, the application maintain a continuous connection with the database throught the entire operation.

* In Connected Architecture is mainly used in real-time applications where live data interaction is required.

Example : 

	1. Banking Applications
	2. Rail way Reservation systems

Example :

Step 1 : Create Table

CREATE TABLE BankAccount
(
	AccountNo Int PRIMARY KEY,
	CustomerName varchar(100),
	AccountType varchar(20),
	Balance decimal(18,2),
	BranchName varchar(100),
	ATMCard bit default 0,
	PassBook bit default 0
);

Step 2 : Create Stored Procedure

CREATE PROCEDURE [dbo].[Usp_BankOperations]
(
	@Action varchar(30),
	@AccountNo int = NULL,
	@CustomerName varchar(100) = NULL,
	@AccountType varchar(20) = NULL,
	@Balance decimal(18,2) = NULL,
	@BranchName varchar(100) = NULL
)
AS
BEGIN
	
	IF @Action = 'OPENACCOUNT'
	BEGIN
	INSERT INTO BankAccount(AccountNo, CustomerName, AccountType, Balance, BranchName)
	VALUES
	(@AccountNo, @CustomerName, @AccountType, @Balance, @BranchName);
	END

	ELSE IF @Action = 'DEPOSIT'
	BEGIN
	UPDATE BankAccount SET Balance = Balance + @Balance WHERE AccountNo = @AccountNo;
	END

	ELSE IF @Action = 'WITHDRAW'
	BEGIN
	UPDATE BankAccount SET Balance = Balance - @Balance WHERE AccountNo = @AccountNo;
	END

	ELSE IF @Action = 'BALANCE'
	BEGIN
	SELECT Balance FROM BankAccount WHERE AccountNo = @AccountNo;
	END

	ELSE IF @Action = 'ACCOUNTDETAILS'
	BEGIN
	SELECT * FROM BankAccount WHERE AccountNo = @AccountNo;
	END

	ELSE IF @Action = 'APPLYATM'
	BEGIN
	UPDATE BankAccount SET ATMCard = 1 WHERE AccountNo = @AccountNo;
	END

	ELSE IF @Action = 'APPLYPASSBOOK'
	BEGIN
	UPDATE BankAccount SET PassBook = 1 WHERE AccountNo = @AccountNo;
	END

	ELSE IF @Action = 'UPDATECUSTOMER'
	BEGIN
	UPDATE BankAccount SET CustomerName = @CustomerName, BranchName = @BranchName WHERE AccountNo = @AccountNo;
	END

	ELSE IF @Action = 'CLOSEACCOUNT'
	BEGIN
	DELETE FROM BankAccount WHERE AccountNo = @AccountNo;
	END

	ELSE IF @Action = 'GETALL'
	BEGIN
	SELECT * FROM BankAccount;
	END
END;
