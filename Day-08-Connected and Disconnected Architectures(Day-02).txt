        private static void AccountDetails()
        {
            SqlConnection conn = new SqlConnection(cs);

            conn.Open();

            SqlCommand cmd = new SqlCommand("Usp_BankOperations", conn);

            cmd.CommandType = CommandType.StoredProcedure;

            Console.Write("Account No : ");
            int accNo = Convert.ToInt32(Console.ReadLine());

            cmd.Parameters.AddWithValue("@Action", "ACCOUNTDETAILS");

            cmd.Parameters.AddWithValue("@AccountNo", accNo);

            SqlDataReader reader = cmd.ExecuteReader();

            while (reader.Read())
            {
                Console.WriteLine("\nAccount No : " + reader["AccountNo"]);
                Console.WriteLine("Customer : " + reader["CustomerName"]);
                Console.WriteLine("Type : " + reader["AccountType"]);
                Console.WriteLine("Balance : " + reader["Balance"]);
                Console.WriteLine("Branch : " + reader["BranchName"]);
                Console.WriteLine("ATM Card : " + reader["ATMCard"]);
                Console.WriteLine("PassBook : " + reader["PassBook"]);
            }

            reader.Close();
            conn.Close();
        }
