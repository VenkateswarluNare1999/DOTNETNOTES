LEFT Join In SQL Server
-----------------------
* A LEFT JOIN returns all rows from the left table and matching rows from the right table.

* If no matching row exists in the right table, SQL Server fills those columns with NULL values.

Syntax :

SELECT 
	columns 
FROM TableA A 
LEFT JOIN TableB B
	ON A.CommonColumn = B.CommonColumn;
	
1. Departments Table	

CREATE TABLE Departments
(
	DepartmentId Int Primary Key,
	DepartmentName varchar(50),

);

2. Employees Table

CREATE TABLE Employees
(
	EmployeeId Int Primary Key,
	EmployeeName varchar(100),
	DepartmentId Int NULL,
	Salary Decimal(10,2),
	ManagerId Int NULL,
	HireDate Date
);

Question 1 : Display all employees along their department names.

SELECT 
	e.EmployeeId, 
	e.EmployeeName, 
	e.DepartmentId, 
	d.DepartmentName
FROM Employees e
LEFT JOIN Departments d
	ON e.DepartmentId = d.DepartmentId;
	
Question 2 : Find employees who are not assigned to any department.

SELECT 
	e.EmployeeId, 
	e.EmployeeName
FROM Employees e
LEFT JOIN Departments d
	ON e.DepartmentId = d.DepartmentId
WHERE d.DepartmentId IS NULL;

Question 3 : Find departments that don't have any employees

SELECT 
	d.DepartmentId, 
	d.DepartmentName
FROM Departments d
LEFT JOIN Employees e
	ON d.DepartmentId = e.DepartmentId
WHERE e.EmployeeId IS NULL;

Question 4 : Display every department and the number of employees in each department

SELECT 
	d.DepartmentId, 
	d.DepartmentName,
	COUNT(e.EmployeeId) AS EmployeeCount
FROM Departments d
LEFT JOIN Employees e
	ON d.DepartmentId = e.DepartmentId
GROUP BY
d.DepartmentId,
d.DepartmentName;

Question 5 : Find departments having more than one employee

SELECT 
	d.DepartmentId, 
	d.DepartmentName,
	COUNT(e.EmployeeId) AS EmployeeCount
FROM Departments d
LEFT JOIN Employees e
	ON d.DepartmentId = e.DepartmentId
GROUP BY
d.DepartmentId,
d.DepartmentName
HAVING COUNT(e.EmployeeId) > 1;

Question 6 : Find employees whose salary is greater than the average salary of their department.

SELECT
	e.EmployeeId,
	e.EmployeeName,
	e.Salary,
	d.DepartmentName
FROM Employees e 
LEFT JOIN Departments d
	ON e.DepartmentId = d.DepartmentId
WHERE e.Salary >
(
	SELECT AVG(e2.Salary)
	FROM Employees e2
	WHERE e2.DepartmentId = e.DepartmentId
);
