Stored Procedure with Conditional Logic
---------------------------------------
* Stored Procedures can contain control-of-flow statements such as IF...ELSE.

Query : To Check whether an employee exists before returning the employee details.

CREATE PROCEDURE sp_CheckEmployee
@EmpId Int
AS
BEGIN
	IF EXISTS
	(
		SELECT 1
		FROM Employees
		WHERE EmpId = @EmpId
	)
	BEGIN
		SELECT 'Emplopyee Found' AS Message;

		SELECT EmpId, Name, Department, Salary
		FROM Employees
		WHERE EmpId = @EmpId;
	END
	ELSE
	BEGIN
		SELECT 'Employee Not Found' AS Message;
	END;
END;

Execute : EXECUTE sp_CheckEmployee @EmpId = 1;

Stored Procedure with Error Handling
-------------------------------------
CREATE PROCEDURE sp_InsertEmployeeSafe
@EmpId Int,
@Name varchar(50),
@Department varchar(50),
@Salary decimal(10,2)
AS
BEGIN
	BEGIN TRY
		INSERT INTO Employees(EmpId, Name, Department, Salary)
		VALUES
		(@EmpId, @Name,@Department,@Salary);

		SELECT 'Employee Inserted Successfully' AS Message;
	END TRY
	BEGIN CATCH
		SELECT
			ERROR_NUMBER() AS ErrorNumber,
			ERROR_MESSAGE() AS ErrorMessage;
	END CATCH;
END;

Execute : EXECUTE sp_InsertEmployeeSafe @EmpId = 6, @Name = 'Venkatesh', @Department = 'Development', @Salary = 75000;
